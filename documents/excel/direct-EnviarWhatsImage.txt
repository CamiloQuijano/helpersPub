Sub EnviarWhatsImageDirect()
Dim Teléfono As String
Dim Texto As String

For Each Celda In Hoja1.Range("Clientes[TELÉFONO]")

    With Envío
        Teléfono = Celda.Value
        Texto = Celda.Offset(0, 1).Value
        ActiveSheet.Shapes(2).Copy
        
        'indicativo + número + texto, utilizamos  VBA.Replace para reemplazar los espacios por %20 porque para whatsapp el espacio es %20
        mensaje = VBA.Replace("whatsapp://send?phone=" & "57" & Teléfono & "&text=" & Texto, " ", "%20")

        x = ShellExecute(hwnd, "Open", mensaje, &O0, &O0, SW_NORMAL)  'Para obtener el mensaje
        Application.Wait Now + TimeValue("00:00:03")
        
        'Escribe el mensaje y lo envía con Enter.  Tambien envía las imagenes
        SendKeys "^v", True
        Application.Wait (Now + TimeValue("00:00:05"))
        SendKeys "~", True
        
        'Pone un mensaje de enviado en la tabla
        Celda.Offset(0, 2).Value = "ENVIADO"
        
            
    End With
Next Celda


End Sub
